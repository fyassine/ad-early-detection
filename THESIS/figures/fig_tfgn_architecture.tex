% TikZ diagram: Detailed Architecture of the Temporal-First Graph Network (TFGN)
% For inclusion in Chapter 4 (Section 4.5)

\begin{tikzpicture}[
    font=\small\sffamily,
    >=Latex,
    node distance=0.6cm and 0.7cm,
    every node/.style={align=center},
    modulebox/.style={
        draw=black!70,
        fill=white,
        rounded corners=5pt,
        line width=0.8pt,
        inner sep=7pt
    },
    corebox/.style={
        draw=TUMBlue!90,
        fill=TUMBlue!10,
        rounded corners=4pt,
        line width=0.9pt,
        inner sep=5pt,
        font=\footnotesize,
        text=black!90
    },
    ablationbox/.style={
        draw=TUMAccentOrange!90,
        fill=TUMAccentOrange!10,
        dashed,
        rounded corners=4pt,
        line width=0.85pt,
        inner sep=5pt,
        font=\footnotesize,
        text=black!90
    },
    databox/.style={
        draw=black!60,
        fill=white,
        rounded corners=3pt,
        line width=0.6pt,
        inner sep=4pt,
        font=\scriptsize,
        text=black!90
    },
    tensorlabel/.style={
        font=\scriptsize\bfseries,
        text=\fg!95
    },
    flowline/.style={
        ->,
        line width=0.85pt,
        draw=\fg!80
    },
    dashedflow/.style={
        ->,
        dashed,
        line width=0.75pt,
        draw=\fg!70
    },
    losslabel/.style={
        font=\scriptsize\bfseries,
        fill=red!8,
        draw=red!60,
        rounded corners=2pt,
        inner sep=2.5pt,
        text=red!80!black
    }
]

% -------------------------------------------------------------
% INPUT BLOCK (Leftmost column)
% -------------------------------------------------------------
\node[databox, text width=2.6cm] (in_feat) at (0, 2.0) {
    \textbf{Regional Visits}\\
    $X \in \mathbb{R}^{T \times 200 \times 200}$\\
    \textnormal{\tiny BOLD connectivity rows}
};

\node[databox, text width=2.6cm] (in_time) at (0, 0.7) {
    \textbf{Interval Spacing}\\
    $\log(1 + \Delta t)$\\
    \textnormal{\tiny Standardised cadence}
};

\node[databox, text width=2.6cm] (in_adj) at (0, -0.6) {
    \textbf{Baseline Graph}\\
    $A_0 \in \{0, 1\}^{200 \times 200}$\\
    \textnormal{\tiny $k$-NN graph ($k=8$)}
};

\node[databox, text width=2.6cm] (in_cov) at (0, -1.9) {
    \textbf{Demographics}\\
    $[\text{age},\, \text{sex}]$\\
    \textnormal{\tiny Covariate vector}
};

% -------------------------------------------------------------
% STAGE 1: NODE-SHARED LSTM (Core S1)
% -------------------------------------------------------------
\node[corebox, text width=2.7cm, right=0.9cm of in_feat] (stage1) {
    \textbf{1. Node-Shared LSTM}\\
    \textnormal{\scriptsize Shared across 200 nodes}\\
    \textnormal{\scriptsize Hidden $d = 64$, 1 layer}\\
    \textnormal{\tiny Dropout 0.3}
};

\draw[flowline] (in_feat) -- (stage1) node[midway, above=4pt, tensorlabel] {$x_i^{(1:T)}$};
\draw[flowline] (in_time.east) -- ++(0.3,0) |- ($(stage1.west)+(0,-0.2)$);

% -------------------------------------------------------------
% STAGE 2: TEMPORAL SALIENCY GATE (Switchable S2)
% -------------------------------------------------------------
\node[ablationbox, text width=2.8cm, right=1.6cm of stage1] (stage2) {
    \textbf{2. Saliency Gate}\\
    \textnormal{\scriptsize (Switchable: S2)}\\
    \textnormal{\tiny $s_i = \sigma(w^\top \text{LReLU}(W_s h_i))$}\\
    \textnormal{\tiny Residual: $(1 + s_i)\,h_i$}
};

\draw[flowline] (stage1) -- (stage2) node[midway, above=4pt, tensorlabel] {$h_i \in \mathbb{R}^{64}$};

% Saliency Loss constraints
\node[losslabel, above=0.35cm of stage2] (loss_saliency) {
    $\mathcal{L}_{\text{sparse}} + \mathcal{L}_{\text{drift}}$
};
\draw[dashedflow] (stage2) -- (loss_saliency);

% -------------------------------------------------------------
% STAGE 3 & 4: GVAE ENCODER & FUSION (Switchable S1c / S3)
% -------------------------------------------------------------
\node[ablationbox, text width=2.8cm, below=1.3cm of stage2] (gvae_enc) {
    \textbf{3. GVAE Encoder}\\
    \textnormal{\scriptsize (Switchable: S1c)}\\
    \textnormal{\tiny GATv2 on $A_0$ + FiLM$(\mu)$}\\
    \textnormal{\tiny Latent $z_i \sim \mathcal{N}(\mu_i, \sigma_i^2)$}
};

\draw[flowline] (in_adj.east) -- ++(0.35,0) |- (gvae_enc.west);
\draw[flowline] (in_cov.east) -- ++(0.35,0) |- ($(gvae_enc.west)+(0,-0.3)$);

\node[losslabel, below=0.35cm of gvae_enc] (loss_gvae) {
    $\mathcal{L}_{\text{recon}} + \beta\mathcal{L}_{\text{KL}}$
};
\draw[dashedflow] (gvae_enc) -- (loss_gvae);

\node[ablationbox, text width=2.7cm, right=1.1cm of stage2] (fusion) {
    \textbf{4. Fusion Layer}\\
    \textnormal{\scriptsize (Switchable: S3)}\\
    \textnormal{\tiny $\text{LN}(W_u[h_i \,\|\, z_i])$}\\
    \textnormal{\tiny Identity under S1}
};

\draw[flowline] (stage2) -- (fusion) node[midway, above=4pt, tensorlabel] {$h_i$};
\draw[dashedflow] (gvae_enc.east) -- ++(0.55,0) |- ($(fusion.west)+(0,-0.25)$) node[pos=0.25, right=4pt, tensorlabel] {$z_i$};

% -------------------------------------------------------------
% STAGE 5: SPATIAL READOUT / POOLING (Core S1 / S4)
% -------------------------------------------------------------
\node[corebox, text width=2.6cm, right=1.6cm of fusion] (readout) {
    \textbf{5. Spatial Readout}\\
    \textnormal{\scriptsize Mean Pooling (S1)}\\
    \textnormal{\tiny or Sparsemax (S4)}\\
    \textnormal{\scriptsize $\mathbf{z}_{\text{patient}} = \bigoplus_{v=1}^{200} u_v$}
};

\draw[flowline] (fusion) -- (readout) node[midway, above=4pt, tensorlabel] {$u_i \in \mathbb{R}^{64}$};

% -------------------------------------------------------------
% STAGE 6: DUAL-SCORE CLASSIFICATION HEAD (Core S1 / S5)
% -------------------------------------------------------------
\node[corebox, text width=2.7cm, right=1.4cm of readout] (head_main) {
    \textbf{6. Dual-Score Head}\\
    \textnormal{\scriptsize Linear / MLP}\\
    \textnormal{\tiny Emits logit $\hat{y}$}\\
    \textnormal{\tiny Saliency $s_{\text{topo}}$ (S5)}
};

\draw[flowline] (readout) -- (head_main) node[midway, above=4pt, tensorlabel] {$\mathbf{z}_{\text{pat}}$};

% Output & Losses
\node[databox, fill=TUMBlue!15, right=0.8cm of head_main] (out_pred) {
    \textbf{Prediction}\\
    $\hat{y} \in [0, 1]$\\
    \textnormal{\tiny Conversion risk}
};
\draw[flowline] (head_main) -- (out_pred);

\node[losslabel, above=0.35cm of head_main] (loss_bce) {
    $\mathcal{L}_{\text{BCE}} + \lambda_{\text{cent}}\mathcal{L}_{\text{cent}}$
};
\draw[dashedflow] (head_main) -- (loss_bce);

% -------------------------------------------------------------
% STAGE 7: COHORT-ADVERSARY HEAD (Escalation Mitigation)
% -------------------------------------------------------------
\node[ablationbox, text width=2.7cm, below=1.3cm of head_main] (head_adv) {
    \textbf{7. Cohort Adversary}\\
    \textnormal{\scriptsize (Escalation Trigger)}\\
    \textnormal{\tiny GRL + Binary Head}\\
    \textnormal{\tiny ADNI vs DELCODE}
};

\draw[dashedflow] (readout.east) -- ++(0.35,0) |- (head_adv.west);

\node[databox, fill=TUMAccentOrange!15, right=0.8cm of head_adv] (out_cohort) {
    \textbf{Cohort Logit}\\
    $\hat{y}_{\text{cohort}}$\\
    \textnormal{\tiny Target: Chance ($0.5$)}
};
\draw[dashedflow] (head_adv) -- (out_cohort);

% -------------------------------------------------------------
% LEGEND BOX (Bottom-left)
% -------------------------------------------------------------
\node[databox, anchor=south west, fill=white, inner sep=5pt] at (-0.2, -4.5) {
    \textbf{Legend:}\quad
    \tikz[baseline=-0.5ex]{\draw[corebox, minimum width=0.4cm, minimum height=0.25cm] (0,0) rectangle (0.4,0.25);} \textnormal{\scriptsize Core Module (S1)}\quad
    \tikz[baseline=-0.5ex]{\draw[ablationbox, minimum width=0.4cm, minimum height=0.25cm] (0,0) rectangle (0.4,0.25);} \textnormal{\scriptsize Switchable Ablation (S2--S5 / Mitigation)}\quad
    \tikz[baseline=-0.5ex]{\draw[dashedflow] (0,0) -- (0.5,0);} \textnormal{\scriptsize Auxiliary / Loss Path}
};

\end{tikzpicture}
