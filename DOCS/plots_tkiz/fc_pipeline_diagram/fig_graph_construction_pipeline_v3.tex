    },
    inputcard/.style={

        draw=TUMAccentOrange!90,
        fill=TUMAccentOrange!12!\bg
    },
    featurescard/.style={
        basecard,

    },
    topologycard/.style={

% Geometry parameters

% Stage 1: rs-fMRI & Parcellation
\node[inputcard, text width=\cardwidthtierone, minimum height=\tieroneheight, anchor=north west] (card1) at (0, 0) {

};

% Stage 2: ROI Time Series Extraction
\node[signalcard, text width=\cardwidthtierone, minimum height=\tieroneheight, anchor=north west] (card2) at ($(card1.north east) + (\colsep, 0)$) {

    \textnormal{\scriptsize Mean BOLD signal within each parcel:}\\[1pt]

};

% Stage 3: Functional Connectivity Matrix
\node[connectivitycard, text width=\cardwidthtierone, minimum height=\tieroneheight, anchor=north west] (card3) at ($(card2.north east) + (\colsep, 0)$) {

};

% Tier 1 Horizontal Flowlines (Generous 0.75cm gap)
\draw[flowline] (card1.east) -- (card2.west |- card1.east) node[midway, above=2pt, flowlabel] {parcellate};

% TIER 2A: DUAL BRANCH REPRESENTATION (Stages 4a, 4b) & MERGE
% -------------------------------------------------------------
% Midpoint coordinate below Tier 1 (Generous 1.50cm vertical gap below Tier 1 bottom)
\coordinate (tier1_mid) at ($(card2.south) + (0, -1.50cm)$);

% Branch 4a: Regional Node Features (Left)
\node[featurescard, text width=5.8cm, minimum height=2.4cm, anchor=north east] (card4a) at ($(tier1_mid) + (-0.45cm, 0)$) {

\coordinate (y_split) at ($(card3.south) + (0, -0.65cm)$);

\draw[flowline_plain] (pt_split_4b) -- (pt_split_4a) node[midway, above=2pt, flowlabel] {extract FC profile};
\draw[flowline] (pt_split_4a) -- (card4a.north);

% Convergence lines into merge node
