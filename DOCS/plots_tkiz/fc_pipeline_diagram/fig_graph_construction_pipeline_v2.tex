% -------------------------------------------------------------
% STAGE 2: BOLD TIME SERIES EXTRACTION
% -------------------------------------------------------------
\node[card2, text width=13.5cm, below=0.7cm of card1] (card2) {

        \textbf{2. Regional BOLD Signal Averaging}\newline
        \textnormal{\footnotesize Mean timecourse per parcel $j \in \{1, \dots, 200\}$: $\displaystyle \mathbf{T} = \begin{bmatrix} \mathbf{t}_1^\top \\ \vdots \\ \mathbf{t}_{200}^\top \end{bmatrix} \in \mathbb{R}^{200 \times L}$, where $L$ is the number of temporal frames per scan.}
    \end{tabular}
};

% -------------------------------------------------------------
% STAGE 3: FUNCTIONAL CONNECTIVITY & FISHER-Z
% -------------------------------------------------------------
\node[card3, text width=13.5cm, below=0.7cm of card2] (card3) {
    \begin{tabular}{@{}m{2.6cm}@{\hspace{7pt}}m{9.9cm}@{}}
        \insetimg{2.5cm}{fc_matrix} &
        \textbf{3. Pairwise Functional Connectivity Matrix \& Fisher $z$-Transformation}\newline
        \textnormal{\footnotesize Pearson correlation $r_{ij} = \operatorname{corr}(\mathbf{t}_i, \mathbf{t}_j)$, Fisher-transformed: $\displaystyle z_{ij} = \operatorname{arctanh}(r_{ij}) = \frac{1}{2}\ln\left(\frac{1+r_{ij}}{1-r_{ij}}\right)$, yielding symmetric $C \in \mathbb{R}^{200 \times 200}$ (zero diagonal).}
    \end{tabular}
};
\draw[flowline] (card2.south) -- (card3.north) node[midway, right=4pt, flowlabel] {Pairwise temporal coupling};

        \textnormal{\footnotesize Each region carries its full connectivity row: $\mathbf{x}_j = C_{j,:} \in \mathbb{R}^{200}$, so $X = C \in \mathbb{R}^{200 \times 200}$.}
