% ── Argument-structure annotations ───────────────────────────────────────────
% Four callout kinds, each taking {short register label}{body}. Every one is
% also written to \jobname.ann, which Appendix E reads back as the Reviewer
% Risk Register — so the register is assembled by the compiler and cannot drift
% out of sync with the prose the way a hand-maintained list would.
%
% \@starttoc/\addcontentsline are the stock LaTeX list-of machinery (the same
% pair behind \listoffigures); using them keeps this dependency-free rather
% than pulling in tcolorbox, which tectonic would then have to fetch.
\newcounter{annotation}
\makeatletter
\newcommand{\listofannotations}{\@starttoc{ann}}
\newcommand{\anno@box}[3]{%
  \refstepcounter{annotation}%
  \addcontentsline{ann}{figure}{\protect\numberline{\theannotation}\textbf{#1} --- #2}%
  \par\vspace{0.6\baselineskip}%
  \noindent\fbox{%
    \parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{%
      \textbf{#1 \theannotation\ · #2}\par\vspace{0.25\baselineskip}#3%
    }%
  }%
  \par\vspace{0.6\baselineskip}%
}
% These four must be defined while '@' is still a letter, since their bodies
% reference \anno@box.
% Open problem in the field — nobody has settled this yet.
\newcommand{\gap}[2]{\anno@box{RESEARCH GAP}{#1}{#2}}
% Contested / unsettled — competent people currently disagree.
\newcommand{\grey}[2]{\anno@box{GREY AREA}{#1}{#2}}
% Where this thesis is defensibly ahead of the published state of the art.
\newcommand{\edge}[2]{\anno@box{OUR EDGE}{#1}{#2}}
% Where this thesis is exposed — what a hostile reviewer attacks first.
\newcommand{\risk}[2]{\anno@box{EXPOSURE}{#1}{#2}}
\makeatother
